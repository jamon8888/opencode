# GDPR MCP Server Implementation Plan

> **For agentic workers:** REQUIRED: Use superpowers:subagent-driven-development (if subagents available) or superpowers:executing-plans to implement this plan. Steps use checkbox (`- [x]`) syntax for tracking.

**Goal:** Build a production-ready Rust MCP server that anonymizes documents (PDF, DOCX, images) through CloakPipe L1+L2 PII detection before any content reaches an LLM, with an immutable ClickHouse audit trail for GDPR/AI Act compliance.

**Architecture:** The server embeds `cloakpipe-core` as a library for in-process PII detection (no network call for the critical path), calls `kreuzberg` as a library for document extraction (75+ formats), persists pseudonym mappings in SQLite, stores anonymized chunks in Qdrant, and writes every operation to ClickHouse as an immutable audit record. Circuit breakers + bulkheads protect every upstream call; CloakPipe failure hard-fails (422) while GLiNER failure degrades gracefully.

**Tech Stack:**
- Rust 1.91+, axum 0.8, tokio 1.50, rmcp 0.1
- `cloakpipe-core` (git dep) — Detector, Vault, Replacer, Rehydrator
- `kreuzberg` 4.x (crates.io) — extract_file, extract_bytes, ExtractionConfig
- rusqlite 0.31 (bundled) — pseudonym→doc_id audit map
- clickhouse 0.12 — immutable audit trail
- qdrant-client — anonymized vector chunks
- prometheus 0.13 — metrics

```rust
use cloakpipe_core::{
    config::{CloakPipeConfig, DetectionConfig, NerConfig, VaultConfig, ResolverConfig},
    detector::Detector,
    replacer::Replacer,
    rehydrator::Rehydrator,
    vault::Vault,
    detector::DetectedEntity,
};
use anyhow::Result;
use std::sync::{Arc, Mutex};

pub struct PiiEngine {
    /// L1 — fast pattern detection. INVARIANT I1: never skipped.
    l1: Detector,
    /// L2 — GLiNER NER. INVARIANT I2: optional/degradable.
    l2: Option<Detector>,
    /// Shared vault (AES-256-GCM). Same entity → same token across L1 and L2.
    vault: Arc<Mutex<Vault>>,
}

pub struct AnonymizeResult {
    pub text:         String,
    pub entities:     Vec<DetectedEntity>,
    pub ner_degraded: bool,
    pub pii_count:    usize,
}

impl PiiEngine {
    /// Production loader: reads vault key from env, loads both L1 and L2 configs.
    pub fn load(l1_config_path: &str, l2_config_path: Option<&str>) -> Result<Self> {
        let key = std::env::var("CLOAKPIPE_VAULT_KEY")
            .map_err(|_| anyhow::anyhow!("CLOAKPIPE_VAULT_KEY must be set"))?
            .into_bytes();

        let l1_cfg = Self::l1_detection_config();
        let l1 = Detector::from_config(&l1_cfg)?;

        let l2 = if let Some(_path) = l2_config_path {
            let l2_cfg = Self::l2_detection_config();
            Detector::from_config(&l2_cfg).ok() // degradable
        } else {
            None
        };

        // Vault path from the L1 config (both layers share same vault file)
        let vault = Vault::open(l1_config_path, key)?;
        Ok(Self { l1, l2, vault: Arc::new(Mutex::new(vault)) })
    }

    /// Test loader: in-memory ephemeral vault, no env vars needed.
    pub fn load_for_test(vault_path: &str) -> Result<Self> {
        let key = std::env::var("CLOAKPIPE_VAULT_KEY")
            .unwrap_or_else(|_| "test-key-32-bytes-long-padding!!".to_string())
            .into_bytes();
        let l1_cfg = Self::l1_detection_config();
        let l1 = Detector::from_config(&l1_cfg)?;
        let vault = Vault::open(vault_path, key)?;
        Ok(Self { l1, l2: None, vault: Arc::new(Mutex::new(vault)) })
    }

    fn l1_detection_config() -> DetectionConfig {
        DetectionConfig {
            secrets:       true,
            financial:     true,
            emails:        true,
            phone_numbers: true,
            dates:         false,
            ip_addresses:  true,
            urls_internal: false,
            ner: NerConfig { enabled: false, ..Default::default() },
            resolver: ResolverConfig { enabled: true, threshold: 0.90 },
            ..Default::default()
        }
    }

    fn l2_detection_config() -> DetectionConfig {
        DetectionConfig {
            secrets:       false,
            financial:     false,
            emails:        false,
            phone_numbers: false,
            ner: NerConfig {
                enabled:              true,
                backend:              cloakpipe_core::config::NerBackend::Gliner,
                model:                Some("/models/gliner".to_string()),
                confidence_threshold: 0.75,
                entity_types: vec![
                    "person name".into(), "full name".into(),
                    "postal address".into(), "national id".into(),
                    "client name".into(), "attorney name".into(),
                    "medical condition".into(), "diagnosis".into(),
                ],
            },
            ..Default::default()
        }
    }

    /// L1 → L2 pipeline. Returns anonymized text + metadata.
    /// INVARIANT I1: if L1 fails, this returns Err (caller must 422).
    /// INVARIANT I2: if L2 unavailable, sets ner_degraded=true and continues.
    pub fn anonymize(&mut self, text: &str, ner_degraded: &mut bool) -> anyhow::Result<AnonymizeResult> {
        let l1_entities = self.l1.detect(text)?;  // I1: propagates error
        let mut vault = self.vault.lock().map_err(|e| anyhow::anyhow!("vault lock: {e}"))?;
        let l1_result = Replacer::pseudonymize(text, &l1_entities, &mut *vault)?;

        let (final_text, all_entities) = match &self.l2 {
            Some(l2) => {
                match l2.detect(&l1_result.text) {
                    Ok(l2_entities) if !l2_entities.is_empty() => {
                        let l2_result = Replacer::pseudonymize(&l1_result.text, &l2_entities, &mut *vault)?;
                        let combined = [l1_entities.clone(), l2_entities].concat();
                        (l2_result.text, combined)
                    }
                    _ => (l1_result.text.clone(), l1_entities.clone()),
                }
            }
            None => {
                *ner_degraded = true;
                (l1_result.text.clone(), l1_entities.clone())
            }
        };

        vault.save()?;
        let pii_count = all_entities.len();
        Ok(AnonymizeResult { text: final_text, entities: all_entities, ner_degraded: *ner_degraded, pii_count })
    }

    /// Rehydrate pseudonyms → originals via Vault.
    pub fn rehydrate(&self, text: &str) -> anyhow::Result<String> {
        let vault = self.vault.lock().map_err(|e| anyhow::anyhow!("vault lock: {e}"))?;
        let result = Rehydrator::rehydrate(text, &*vault)?;
        Ok(result.text)
    }

    /// Vault stats for metrics/audit.
    pub fn vault_stats(&self) -> anyhow::Result<cloakpipe_core::vault::VaultStats> {
        let vault = self.vault.lock().map_err(|e| anyhow::anyhow!("vault lock: {e}"))?;
        Ok(vault.stats())
    }
}
```

```rust
use kreuzberg::{
    extract_file, extract_bytes as kreuzberg_extract_bytes,
    ExtractionConfig, OcrConfig, ExtractionResult,
};
use std::path::Path;
use anyhow::Result;

/// Build ExtractionConfig for GDPR document ingest.
/// OCR via Tesseract is always available; quality processing is on.
pub fn ingest_config(language: &str) -> ExtractionConfig {
    ExtractionConfig {
        use_cache: true,
        enable_quality_processing: true,
        force_ocr: false,
        ocr: Some(OcrConfig {
            backend: "tesseract".to_string(),
            language: lang_to_tesseract(language),
            ..Default::default()
        }),
        ..Default::default()
    }
}

/// Extract text from a document path.
/// Emits a tracing warning if quality_score < 0.4.
pub async fn extract_document(path: &Path, language: &str) -> Result<ExtractionResult> {
    let config = ingest_config(language);
    let result = extract_file(path, None, Some(config)).await
        .map_err(|e| anyhow::anyhow!("kreuzberg extract_file: {e}"))?;

    if result.quality_score < 0.4 {
        tracing::warn!(
            path = %path.display(),
            quality = result.quality_score,
            "Low extraction quality — consider force_ocr=true"
        );
    }
    Ok(result)
}

/// Extract from raw bytes (uploaded file not on disk).
pub async fn extract_document_bytes(
    data: &[u8],
    mime_type: &str,
    language: &str,
) -> Result<ExtractionResult> {
    let config = ingest_config(language);
    kreuzberg_extract_bytes(data.to_vec(), mime_type.to_string(), None, Some(config))
        .await
        .map_err(|e| anyhow::anyhow!("kreuzberg extract_bytes: {e}"))
}

fn lang_to_tesseract(lang: &str) -> String {
    match lang {
        "fr" => "fra".to_string(),
        "de" => "deu".to_string(),
        "es" => "spa".to_string(),
        "it" => "ita".to_string(),
        "nl" => "nld".to_string(),
        "pt" => "por".to_string(),
        _ => "eng+fra+deu".to_string(),
    }
}
```

```rust
use std::sync::{Arc, Mutex};
use anyhow::Result;

#[derive(Debug, Clone)]
pub struct DocEntityRow {
    pub entity_type:     String,
    pub pseudonym:       String,
    pub detection_layer: String, // "l1_pattern" | "l2_ner"
    pub confidence:      Option<f64>,
    pub ner_degraded:    bool,
}

#[derive(Debug)]
pub struct DocumentSummary {
    pub document_id:  String,
    pub entity_count: usize,
}

pub struct DocAuditDb {
    conn: Arc<Mutex<Connection>>,
}

impl DocAuditDb {
    pub fn open(path: &str) -> Result<Self> {
        let conn = Connection::open(path)?;
        conn.execute_batch("
            CREATE TABLE IF NOT EXISTS doc_entity_map (
                id              TEXT PRIMARY KEY DEFAULT (lower(hex(randomblob(8)))),
                document_id     TEXT NOT NULL,
                entity_type     TEXT NOT NULL,
                pseudonym       TEXT NOT NULL,
                detection_layer TEXT NOT NULL,
                confidence      REAL,
                ner_degraded    INTEGER NOT NULL DEFAULT 0,
                created_at      TEXT NOT NULL DEFAULT (datetime('now'))
            );
            CREATE INDEX IF NOT EXISTS idx_document_id ON doc_entity_map(document_id);
        ")?;
        Ok(Self { conn: Arc::new(Mutex::new(conn)) })
    }

    pub fn record_entities(&self, doc_id: &str, rows: &[DocEntityRow]) -> Result<()> {
        let conn = self.conn.lock().unwrap();
        for row in rows {
            conn.execute(
                "INSERT INTO doc_entity_map (document_id, entity_type, pseudonym, detection_layer, confidence, ner_degraded)
                 VALUES (?1, ?2, ?3, ?4, ?5, ?6)",
                params![doc_id, row.entity_type, row.pseudonym, row.detection_layer,
                        row.confidence, row.ner_degraded as i32],
            )?;
        }
        Ok(())
    }

    /// GDPR Art. 17: remove all pseudonym→doc mappings. Returns rows deleted.
    pub fn delete_document(&self, doc_id: &str) -> Result<usize> {
        let conn = self.conn.lock().unwrap();
        let n = conn.execute("DELETE FROM doc_entity_map WHERE document_id = ?1", params![doc_id])?;
        Ok(n)
    }

    pub fn entities_for_doc(&self, doc_id: &str) -> Result<Vec<DocEntityRow>> {
        let conn = self.conn.lock().unwrap();
        let mut stmt = conn.prepare(
            "SELECT entity_type, pseudonym, detection_layer, confidence, ner_degraded
             FROM doc_entity_map WHERE document_id = ?1"
        )?;
        let rows = stmt.query_map(params![doc_id], |r| Ok(DocEntityRow {
            entity_type:     r.get(0)?,
            pseudonym:       r.get(1)?,
            detection_layer: r.get(2)?,
            confidence:      r.get(3)?,
            ner_degraded:    r.get::<_, i32>(4)? != 0,
        }))?.collect::<Result<Vec<_>, _>>()?;
        Ok(rows)
    }

    pub fn list_documents(&self) -> Result<Vec<DocumentSummary>> {
        let conn = self.conn.lock().unwrap();
        let mut stmt = conn.prepare(
            "SELECT document_id, COUNT(*) as cnt FROM doc_entity_map GROUP BY document_id"
        )?;
        let docs = stmt.query_map([], |r| Ok(DocumentSummary {
            document_id:  r.get(0)?,
            entity_count: r.get::<_, i64>(1)? as usize,
        }))?.collect::<Result<Vec<_>, _>>()?;
        Ok(docs)
    }
}
```

```rust
use std::sync::atomic::{AtomicU8, AtomicU64, Ordering};
use std::sync::Arc;
use std::time::{Duration, SystemTime, UNIX_EPOCH};
use tokio::time::timeout;
use anyhow::Result;

const CLOSED: u8 = 0;
const OPEN: u8 = 1;
const HALF_OPEN: u8 = 2;

pub struct CircuitBreaker {
    name:            &'static str,
    state:           AtomicU8,
    failure_count:   AtomicU64,
    failure_threshold: u64,
    open_at_secs:    AtomicU64,
    recovery_secs:   u64,
}

impl CircuitBreaker {
    pub fn new(name: &'static str, failure_threshold: u64, recovery: Duration) -> Arc<Self> {
        Arc::new(Self {
            name,
            state: AtomicU8::new(CLOSED),
            failure_count: AtomicU64::new(0),
            failure_threshold,
            open_at_secs: AtomicU64::new(0),
            recovery_secs: recovery.as_secs(),
        })
    }

    pub fn is_open(&self) -> bool {
        match self.state.load(Ordering::SeqCst) {
            OPEN => {
                let now = now_secs();
                let open_at = self.open_at_secs.load(Ordering::SeqCst);
                if now - open_at >= self.recovery_secs {
                    self.state.store(HALF_OPEN, Ordering::SeqCst);
                    false // allow half-open probe
                } else {
                    true
                }
            }
            _ => false,
        }
    }

    pub fn record_failure(&self) {
        let count = self.failure_count.fetch_add(1, Ordering::SeqCst) + 1;
        if count >= self.failure_threshold {
            self.state.store(OPEN, Ordering::SeqCst);
            self.open_at_secs.store(now_secs(), Ordering::SeqCst);
        }
    }

    pub fn record_success(&self) {
        self.failure_count.store(0, Ordering::SeqCst);
        self.state.store(CLOSED, Ordering::SeqCst);
    }

    pub async fn call<F, T>(&self, fut: F) -> Result<T>
    where F: std::future::Future<Output = Result<T>>,
    {
        if self.is_open() {
            return Err(anyhow::anyhow!("circuit breaker open: {}", self.name));
        }
        match fut.await {
            Ok(v)  => { self.record_success(); Ok(v) }
            Err(e) => { self.record_failure(); Err(e) }
        }
    }
}

fn now_secs() -> u64 {
    SystemTime::now().duration_since(UNIX_EPOCH).unwrap_or_default().as_secs()
}
```

```rust
pub struct UpstreamGuard {

impl UpstreamGuard {
    pub fn new(
        name:      &'static str,
        cb_threshold: u64,
        cb_recovery: Duration,

    ) -> Self {
        Self {

            bh: Bulkhead::new(name, bh_max),
            timeout_secs: 30,
        }
    }

    pub fn with_timeout(mut self, secs: u64) -> Self {
        self.timeout_secs = secs;
        self
    }

    where
        F: FnOnce() -> Fut,
        Fut: std::future::Future<Output = Result<T>>,
    {
        let _permit = self.bh.try_acquire()?;
        self.cb.call(
            timeout(Duration::from_secs(self.timeout_secs), f())
                .await
                .map_err(|_| anyhow::anyhow!("upstream timeout"))?
        ).await
    }
```

```rust
use crate::state::AppState;
use crate::extraction::extract_document;
use crate::audit::DocEntityRow;
use crate::error::{GdprError, Result};
use serde::Serialize;
use std::path::Path;
use uuid::Uuid;

#[derive(Serialize)]
pub struct IngestResult {
    pub document_id:        String,
    pub pii_count:          usize,
    pub ner_degraded:       bool,
    pub anonymized_preview: String,  // first 500 chars of anonymized text
    pub quality_score:      f64,
    pub detected_languages: Vec<String>,
}

pub async fn ingest_document(
    state:       &AppState,
    path:        String,
    language:    String,
    legal_basis: String,
) -> Result<IngestResult> {
    // 1. Extract raw text via kreuzberg
    let extracted = extract_document(Path::new(&path), &language)
        .await
        .map_err(|e| GdprError::Extraction(e.to_string()))?;

    // 2. Anonymize via CloakPipe L1+L2
    let mut ner_degraded = false;
    let anon_result = {
        let mut engine = state.engine.lock()
            .map_err(|_| GdprError::Storage("engine lock".into()))?;
        engine.anonymize(&extracted.content, &mut ner_degraded)
            .map_err(GdprError::PiiDetection)?
    };

    // 3. Generate document ID
    let doc_id = Uuid::new_v4().to_string();

    // 4. Record entities in SQLite audit map
    let entity_rows: Vec<DocEntityRow> = anon_result.entities.iter().map(|e| DocEntityRow {
        entity_type:     format!("{:?}", e.category),
        pseudonym:       String::new(), // vault-generated; we don't store originals
        detection_layer: "l1_pattern".to_string(),
        confidence:      Some(e.confidence as f64),
        ner_degraded,
    }).collect();
    state.audit_db.record_entities(&doc_id, &entity_rows)
        .map_err(|e| GdprError::Storage(e.to_string()))?;

    // 5. Preview (no raw PII)
    let preview: String = anon_result.text.chars().take(500).collect();

    Ok(IngestResult {
        document_id:        doc_id,
        pii_count:          anon_result.pii_count,
        ner_degraded:       anon_result.ner_degraded,
        anonymized_preview: preview,
        quality_score:      extracted.quality_score,
        detected_languages: extracted.detected_languages.unwrap_or_default(),
    })
}
```
